Complete Copy-and-Paste HTML Starter

Save this snippet as index.html and open it directly in any browser: 

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Table with CDN</title>
  <!-- Bootstrap 5 CSS -->
  <link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.3/dist/css/bootstrap.min.css" rel="stylesheet">
</head>
<body class="bg-light p-4">
  <div class="container">
    <h2 class="mb-3">Declarative Table via CDN</h2>
    <div id="my-table-container"></div>
  </div>

  <!-- Load V55 Bundle via Official CDN -->
  <script type="module" src="https://keshavsoft.github.io/json-renderers-build-table/dist/v55/min.js"></script>

  <!-- Application Logic -->
  <script type="module">
    const columns = [
      { columnName: "itemName", label: "Item Name" },
      { columnName: "category", label: "Category" },
      { columnName: "baseUnit", label: "Unit" }
    ];

    const data = [
      { itemName: "High Tensile Bolt", category: "Fasteners", baseUnit: "kgs" },
      { itemName: "Galvanized Wire Coil", category: "Raw Materials", baseUnit: "kgs" }
    ];

    // STEP 1: Pure Rendering via Global Registration
    window.ks.jsonRenderers.build.table.render({
      data,
      columns,
      targetHtmlId: "my-table-container",
      type: "withSearchAndFooter",
      options: { showSerial: true, showOptions: true }
    });

    // STEP 2: Hook Domain Events
    window.ks.jsonRenderers.build.table.hookEvents({
      container: "my-table-container",
      dataset: data,
      columns,
      onSearch: (q, filtered) => console.log("Search query:", q),
      onRowClick: (row) => alert("Selected: " + row.itemName),
      onSave: (newRow) => console.log("Added row:", newRow)
    });
  </script>
</body>
</html>
